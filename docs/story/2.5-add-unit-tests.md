## 1. Objective

Establish a test suite that validates the core CV pipeline modules without requiring a real YOLO model or video files. The tests must:

- Cover the segmenter fully with pure data tests (no mocking needed).
- Cover the detector and cropper with mocks for YOLO, OpenCV, and MoviePy.
- Cover the pipeline orchestrator with mocked services.
- Run fast (<10s total) in CI without GPU or model weights.

### 4.3 `tests/test_segmenter.py`

import pytest
from app.services.detector import DetectionResult, VideoMetadata
from app.services.segmenter import (
    identify_segments,
    format_detection_times,
    Segment,
    SegmentationResult,
    _seconds_to_time,
)


class TestIdentifySegments:
    """Tests for identify_segments()."""

    def test_no_detections(self, sample_detection_all_false):
        """Zero detections → zero segments."""
        result = identify_segments(sample_detection_all_false)
        assert len(result.segments) == 0
        assert result.total_seconds_scanned == 30
        assert result.seconds_with_detection == 0

    def test_two_qualifying_segments(self, sample_detection_with_segments):
        """Two segments >= 5s qualify, one short segment filtered out."""
        result = identify_segments(sample_detection_with_segments)

        assert len(result.segments) == 2

        seg0 = result.segments[0]
        assert seg0.index == 0
        assert seg0.start_seconds == 2
        assert seg0.end_seconds == 12
        assert seg0.duration_seconds == 10
        assert seg0.bbox == [100.0, 450.0, 800.0, 120.0]

        seg1 = result.segments[1]
        assert seg1.index == 1
        assert seg1.start_seconds == 22
        assert seg1.end_seconds == 30
        assert seg1.duration_seconds == 8
        assert seg1.bbox == [105.0, 455.0, 795.0, 118.0]

    def test_detection_runs_to_end(self, sample_metadata):
        """Segment that extends to the last second of the video."""
        detections = [False] * 20 + [True] * 10
        bboxes = {20: [50.0, 50.0, 200.0, 100.0]}
        det_result = DetectionResult(
            metadata=sample_metadata, detections=detections, bboxes=bboxes,
        )
        result = identify_segments(det_result)

        assert len(result.segments) == 1
        assert result.segments[0].start_seconds == 20
        assert result.segments[0].end_seconds == 30

    def test_all_detected(self, sample_metadata):
        """Entire video detected → one big segment."""
        detections = [True] * 30
        bboxes = {0: [10.0, 20.0, 300.0, 150.0]}
        det_result = DetectionResult(
            metadata=sample_metadata, detections=detections, bboxes=bboxes,
        )
        result = identify_segments(det_result)

        assert len(result.segments) == 1
        assert result.segments[0].start_seconds == 0
        assert result.segments[0].end_seconds == 30
        assert result.segments[0].duration_seconds == 30

    def test_single_second_detections_filtered(self, sample_metadata):
        """Isolated single-second detections are too short (< 5s)."""
        detections = [False] * 30
        detections[5] = True
        detections[15] = True
        detections[25] = True
        bboxes = {5: [0, 0, 100, 100]}
        det_result = DetectionResult(
            metadata=sample_metadata, detections=detections, bboxes=bboxes,
        )
        result = identify_segments(det_result)

        assert len(result.segments) == 0

    def test_summary_stats(self, sample_detection_with_segments):
        """Verify total_seconds_scanned and seconds_with_detection."""
        result = identify_segments(sample_detection_with_segments)
        assert result.total_seconds_scanned == 30
        # 10 + 4 + 8 = 22 seconds with detection
        assert result.seconds_with_detection == 22


class TestFormatDetectionTimes:
    """Tests for format_detection_times()."""

    def test_empty_segments(self):
        result = format_detection_times([])
        assert result == ""

    def test_formatted_output(self):
        segments = [
            Segment(0, 172, 205, 33, "0:02:52", "0:03:25", [0, 0, 0, 0]),
            Segment(1, 310, 360, 50, "0:05:10", "0:06:00", [0, 0, 0, 0]),
        ]
        result = format_detection_times(segments)
        assert result == "0:02:52 - 0:03:25\n0:05:10 - 0:06:00"


class TestSecondsToTime:
    """Tests for _seconds_to_time()."""

    def test_under_minute(self):
        assert _seconds_to_time(45) == "0:00:45"

    def test_minutes(self):

    def test_hours(self):
        assert _seconds_to_time(3661) == "1:01:01"

import pytest
import numpy as np
from unittest.mock import patch, MagicMock
from app.services.segmenter import Segment
from app.services.cropper import _crop_frame_by_bbox, crop_segments


class TestCropFrameByBbox:
    """Tests for the per-frame bbox cropping function."""

    def test_basic_crop(self):
        frame = np.zeros((480, 640, 3), dtype=np.uint8)
        result = _crop_frame_by_bbox(frame, [100.0, 200.0, 400.0, 100.0])
        assert result.shape == (100, 400, 3)

    def test_even_dimension_enforcement(self):
        """Odd width/height get decremented to even numbers."""
        frame = np.zeros((480, 640, 3), dtype=np.uint8)
        result = _crop_frame_by_bbox(frame, [100.0, 200.0, 401.0, 101.0])
        # 401 → 400, 101 → 100
        assert result.shape[1] <= 400  # width
        assert result.shape[0] <= 100  # height
        assert result.shape[1] % 2 == 0
        assert result.shape[0] % 2 == 0

    def test_clamp_to_frame_bounds(self):
        """Bbox exceeding frame dimensions gets clamped."""
        frame = np.zeros((480, 640, 3), dtype=np.uint8)
        result = _crop_frame_by_bbox(frame, [500.0, 400.0, 400.0, 200.0])
        # x_min=500, w=400 → x_max=min(640, 900)=640 → actual w=140
        # y_min=400, h=200 → y_max=min(480, 600)=480 → actual h=80
        assert result.shape[1] <= 640
        assert result.shape[0] <= 480

    def test_negative_origin_clamped(self):
        """Negative x_min/y_min get clamped to 0."""
        frame = np.zeros((480, 640, 3), dtype=np.uint8)
        result = _crop_frame_by_bbox(frame, [-50.0, -30.0, 200.0, 100.0])
        # x_min clamped to 0, y_min clamped to 0
        assert result.shape[0] > 0
        assert result.shape[1] > 0


class TestCropSegments:
    """Tests for the full segment export flow (mocked MoviePy)."""

    @patch("app.services.cropper.VideoFileClip")
    def test_exports_correct_number_of_files(self, mock_vfc_class, tmp_path):
        """Each segment produces one output file call."""
        mock_video = MagicMock()
        mock_video.duration = 60.0
        mock_subclip = MagicMock()
        mock_cropped = MagicMock()
        mock_video.subclipped.return_value = mock_subclip
        mock_subclip.image_transform.return_value = mock_cropped
        mock_vfc_class.return_value = mock_video

        segments = [
            Segment(0, 10, 20, 10, "0:00:10", "0:00:20", [100, 200, 400, 100]),
            Segment(1, 30, 45, 15, "0:00:30", "0:00:45", [110, 210, 390, 110]),
        ]

        result = crop_segments("test.mp4", segments, str(tmp_path))

        assert len(result.segments) == 2
        assert result.segments[0].file_name == "segment_000000.mp4"
        assert result.segments[1].file_name == "segment_000001.mp4"
        # write_videofile should have been called twice
        assert mock_cropped.write_videofile.call_count == 2

    @patch("app.services.cropper.VideoFileClip")
    def test_empty_segments_returns_empty_result(self, mock_vfc_class, tmp_path):
        """Zero segments → zero exports, no MoviePy calls."""
        mock_video = MagicMock()
        mock_video.duration = 60.0
        mock_vfc_class.return_value = mock_video

        result = crop_segments("test.mp4", [], str(tmp_path))

        assert len(result.segments) == 0

    @patch("app.services.cropper.VideoFileClip")
    def test_segment_failure_continues(self, mock_vfc_class, tmp_path):
        """If one segment export fails, others still proceed."""
        mock_video = MagicMock()
        mock_video.duration = 60.0

        call_count = 0
        def subclipped_side_effect(start, end):
            nonlocal call_count
            call_count += 1
            if call_count == 1:
                raise RuntimeError("FFmpeg exploded")
            mock_sub = MagicMock()
            mock_cropped = MagicMock()

        mock_vfc_class.return_value = mock_video

        segments = [
            Segment(0, 5, 15, 10, "0:00:05", "0:00:15", [100, 200, 400, 100]),
            Segment(1, 20, 30, 10, "0:00:20", "0:00:30", [110, 210, 390, 110]),
        ]

        result = crop_segments("test.mp4", segments, str(tmp_path))

        # Segment 0 failed, segment 1 succeeded
        assert len(result.segments) == 1
        assert result.segments[0].index == 1

import pytest
from unittest.mock import patch, MagicMock
from app.services.detector import DetectionResult, VideoMetadata
from app.services.segmenter import SegmentationResult, Segment
from app.services.cropper import CropResult, CroppedSegment
from app.services.pipeline import run_pipeline, PipelineError, PipelinePhase


@pytest.fixture
def mock_services():
    """Patch all three services and return their mocks."""
    with (
        patch("app.services.pipeline.get_detector") as mock_get_det,
        patch("app.services.pipeline.identify_segments") as mock_seg,
        patch("app.services.pipeline.crop_segments") as mock_crop,
        patch("app.services.pipeline.format_detection_times") as mock_fmt,
        patch("app.services.pipeline.settings") as mock_settings,
    ):
        mock_settings.output_dir = "storage/outputs"

        # Default detector return
        mock_detector = MagicMock()
        mock_get_det.return_value = mock_detector
        metadata = VideoMetadata(fps=30.0, frame_count=900, duration_seconds=30.0)
        det_result = DetectionResult(
            metadata=metadata, detections=[True] * 30, bboxes={0: [10, 20, 100, 50]},
        )
        mock_detector.detect.return_value = det_result

        # Default segmenter return
        segment = Segment(0, 0, 30, 30, "0:00:00", "0:00:30", [10, 20, 100, 50])
        seg_result = SegmentationResult(
            segments=[segment], total_seconds_scanned=30, seconds_with_detection=30,
        )
        mock_seg.return_value = seg_result

        # Default cropper return
        cropped = CroppedSegment(0, "/out/segment_000000.mp4", "segment_000000.mp4", 0, 30, 30, [10, 20, 100, 50])
        crop_result = CropResult(segments=[cropped], output_dir="/out")
        mock_crop.return_value = crop_result

        # Detection times
        mock_fmt.return_value = "0:00:00 - 0:00:30"

        yield {
            "detector": mock_detector,
            "segmenter": mock_seg,
            "cropper": mock_crop,
            "format": mock_fmt,
        }


class TestRunPipeline:
    """Tests for the pipeline orchestrator."""

    def test_calls_all_phases_in_order(self, mock_services, tmp_path):
        with patch("app.services.pipeline.settings") as ms:
            ms.output_dir = str(tmp_path)
            result = run_pipeline("job-1", "video.mp4")

        mock_services["detector"].detect.assert_called_once()
        mock_services["segmenter"].assert_called_once()
        mock_services["cropper"].assert_called_once()
        assert result.job_id == "job-1"

    def test_progress_callback_called(self, mock_services, tmp_path):
        progress_calls = []

        def on_progress(phase, current, total):
            progress_calls.append((phase, current, total))

        with patch("app.services.pipeline.settings") as ms:
            ms.output_dir = str(tmp_path)
            run_pipeline("job-2", "video.mp4", on_progress=on_progress)

        # Should have at least segmentation 0/1 and 1/1
        phases_seen = {call[0] for call in progress_calls}
        assert PipelinePhase.SEGMENTATION in phases_seen

    def test_detection_error_raises_pipeline_error(self, mock_services, tmp_path):
        mock_services["detector"].detect.side_effect = RuntimeError("model failed")

        with patch("app.services.pipeline.settings") as ms:
            ms.output_dir = str(tmp_path)
            with pytest.raises(PipelineError) as exc_info:
                run_pipeline("job-3", "video.mp4")

        assert exc_info.value.phase == PipelinePhase.DETECTION
        assert "model failed" in str(exc_info.value)

    def test_zero_segments_skips_cropping(self, mock_services, tmp_path):
        mock_services["segmenter"].return_value = SegmentationResult(
            segments=[], total_seconds_scanned=30, seconds_with_detection=0,
        )

        with patch("app.services.pipeline.settings") as ms:
            ms.output_dir = str(tmp_path)
            result = run_pipeline("job-4", "video.mp4")

        mock_services["cropper"].assert_not_called()
        assert len(result.crop.segments) == 0
