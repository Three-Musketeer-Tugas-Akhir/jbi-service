# Story 4.9 — Migrate Job Store to PostgreSQL

## Description

Replace the in-memory `JobStore` (dict-based) with a PostgreSQL-backed persistence layer using SQLAlchemy (async) so that job data survives server restarts and can be shared across multiple worker processes.

**Database connection:**
- Host: `localhost`
- Port: `5432`
- Database: `anotasi_db`
- Connection string: `postgresql+asyncpg://postgres:postgres@localhost:5432/anotasi_db`

```python
from sqlalchemy.ext.asyncio import create_async_engine, async_sessionmaker, AsyncSession
from app.config import settings

engine = create_async_engine(
    settings.database_url,
    echo=settings.database_echo,
    pool_size=5,
    max_overflow=10,
)

async_session = async_sessionmaker(engine, class_=AsyncSession, expire_on_commit=False)


async def get_db() -> AsyncSession:
    """FastAPI dependency that yields a database session."""
    async with async_session() as session:
        try:
            yield session
            await session.commit()
        except Exception:
            await session.rollback()
            raise
```

### 3.2 `app/models/job_model.py` — SQLAlchemy ORM Model

```python
import uuid
from datetime import datetime, timezone

from sqlalchemy import (
    String, Text, Float, Integer, DateTime, Enum as SAEnum,
)
from sqlalchemy.dialects.postgresql import UUID, JSONB
from sqlalchemy.orm import DeclarativeBase, Mapped, mapped_column


class Base(DeclarativeBase):
    pass


class JobRecord(Base):
    __tablename__ = "jobs"

    id: Mapped[str] = mapped_column(
        UUID(as_uuid=False), primary_key=True, default=lambda: str(uuid.uuid4()),
    )
    status: Mapped[str] = mapped_column(
        String(20), nullable=False, default="queued", index=True,
    )
    video_path: Mapped[str] = mapped_column(Text, nullable=False, default="")
    output_dir: Mapped[str] = mapped_column(Text, nullable=False, default="")

    # Progress (flattened — no need for a separate table)
    progress_phase: Mapped[str] = mapped_column(String(30), nullable=False, default="")
    progress_current: Mapped[int] = mapped_column(Integer, nullable=False, default=0)
    progress_total: Mapped[int] = mapped_column(Integer, nullable=False, default=0)
    progress_percent: Mapped[float] = mapped_column(Float, nullable=False, default=0.0)

    error: Mapped[str | None] = mapped_column(Text, nullable=True)

    # Pipeline result stored as JSON (segments, detection metadata, etc.)
    result_data: Mapped[dict | None] = mapped_column(JSONB, nullable=True)

    created_at: Mapped[datetime] = mapped_column(
        DateTime(timezone=True),
        nullable=False,
        default=lambda: datetime.now(timezone.utc),
    )
    updated_at: Mapped[datetime] = mapped_column(
        DateTime(timezone=True),
        nullable=False,
        default=lambda: datetime.now(timezone.utc),
        onupdate=lambda: datetime.now(timezone.utc),
    )
```

### 3.3 `app/models/job_store.py` — Rewrite with Async SQLAlchemy

Replace the in-memory `JobStore` class with an async DB-backed implementation. Keep the same public interface so callers (routers, workers, cleanup) need minimal changes.

```python
import logging
from datetime import datetime, timezone
from typing import Optional

from sqlalchemy import select, update, delete, func
from sqlalchemy.ext.asyncio import AsyncSession

from app.models.job import JobStatus, JobProgress, can_transition, is_terminal
from app.models.job_model import JobRecord

logger = logging.getLogger(__name__)


class JobStore:
    """PostgreSQL-backed job store with async SQLAlchemy."""

    # --- Create ---
    async def create(self, session: AsyncSession, job_id: str, video_path: str) -> JobRecord:
        record = JobRecord(id=job_id, video_path=video_path)
        session.add(record)
        await session.flush()
        return record

    # --- Read ---
    async def get(self, session: AsyncSession, job_id: str) -> Optional[JobRecord]:
        result = await session.execute(
            select(JobRecord).where(JobRecord.id == job_id)
        )
        return result.scalar_one_or_none()

    # --- Update status ---
    async def update_status(
        self, session: AsyncSession, job_id: str,
        status: JobStatus, error: str = None,
    ) -> bool:
        record = await self.get(session, job_id)
        if not record:
            return False

        current = JobStatus(record.status)
        if not can_transition(current, status):
            logger.warning(
                "Invalid transition for job %s: %s -> %s",
                job_id, current.value, status.value,
            )
            return False

        record.status = status.value
        record.updated_at = datetime.now(timezone.utc)
        if error:
            record.error = error
        await session.flush()
        return True

    # --- Update progress ---
    async def update_progress(
        self, session: AsyncSession, job_id: str,
        phase: str, current: int, total: int,
    ):
        record = await self.get(session, job_id)
        if record and not is_terminal(JobStatus(record.status)):
            record.progress_phase = phase
            record.progress_current = current
            record.progress_total = total
            record.progress_percent = round((current / total * 100) if total > 0 else 0, 1)
            record.updated_at = datetime.now(timezone.utc)
            await session.flush()

    # --- Set result ---
    async def set_result(
        self, session: AsyncSession, job_id: str, result_data: dict,
    ):
        record = await self.get(session, job_id)
        if record:
            record.result_data = result_data
            await session.flush()

    # --- Delete ---
    async def delete(self, session: AsyncSession, job_id: str) -> bool:
        result = await session.execute(
            delete(JobRecord).where(JobRecord.id == job_id)
        )
        return result.rowcount > 0

    # --- List (with optional filter & pagination) ---
    async def list_filtered(
        self, session: AsyncSession,
        status: Optional[str] = None,
        limit: int = 20,
        offset: int = 0,
    ) -> tuple[list[JobRecord], int]:
        """Returns (page_of_records, total_count)."""
        query = select(JobRecord)
        count_query = select(func.count()).select_from(JobRecord)

        if status:
            query = query.where(JobRecord.status == status)
            count_query = count_query.where(JobRecord.status == status)

        query = query.order_by(JobRecord.created_at.desc())
        query = query.limit(limit).offset(offset)

        result = await session.execute(query)
        records = list(result.scalars().all())

        count_result = await session.execute(count_query)
        total = count_result.scalar()

        return records, total

    # --- List all (for cleanup) ---
    async def list_all(self, session: AsyncSession) -> list[JobRecord]:
        result = await session.execute(
            select(JobRecord).order_by(JobRecord.created_at.desc())
        )
        return list(result.scalars().all())


job_store = JobStore()
```

### 5.3 `app/workers/tasks.py` — Use Sync Session in Background Worker

Background tasks run outside the request lifecycle. Use a **sync** session or run async code in a new event loop:

```python
from app.database import async_session

async def _process_job_async(job_id: str, video_path: str):
    async with async_session() as session:
        record = await job_store.get(session, job_id)
        if not record or is_terminal(JobStatus(record.status)):
            return

        await job_store.update_status(session, job_id, JobStatus.PROCESSING)
        await session.commit()

    # Run pipeline (CPU-bound, sync)
    # On each progress callback, open a new session to update progress
    def on_progress(phase, current, total):
        import asyncio
        loop = asyncio.new_event_loop()
        loop.run_until_complete(_update_progress(job_id, phase, current, total))
        loop.close()

    try:
        result = run_pipeline(job_id=job_id, video_path=video_path, on_progress=on_progress)
        result_data = serialize_pipeline_result(result)

        async with async_session() as session:
            await job_store.set_result(session, job_id, result_data)
            await job_store.update_status(session, job_id, JobStatus.COMPLETED)
            await session.commit()

    except PipelineError as exc:
        async with async_session() as session:
            await job_store.update_status(session, job_id, JobStatus.FAILED, error=str(exc))
            await session.commit()


async def _update_progress(job_id, phase, current, total):
    async with async_session() as session:
        await job_store.update_progress(session, job_id, phase, current, total)
        await session.commit()


def process_job(job_id: str, video_path: str):
    """Entry point for BackgroundTasks — bridges sync → async."""
    import asyncio
    asyncio.run(_process_job_async(job_id, video_path))
```

## 6. Result Serialization / Deserialization

Add helper functions to convert between `PipelineResult` dataclass and JSON-serializable dict for `result_data` JSONB column:

### `app/models/result_serializer.py`

```python
from app.services.pipeline import PipelineResult
from app.services.detector import DetectionResult, VideoMetadata
from app.services.segmenter import SegmentationResult, Segment
from app.services.cropper import CropResult, CroppedSegment


def serialize_pipeline_result(r: PipelineResult) -> dict:
    return {
        "video_path": r.video_path,
        "output_dir": r.output_dir,
        "detection_times_path": r.detection_times_path,
        "detection": {
            "metadata": {
                "fps": r.detection.metadata.fps,
                "frame_count": r.detection.metadata.frame_count,
                "duration_seconds": r.detection.metadata.duration_seconds,
            },
            "total_detections": sum(r.detection.detections),
            "total_seconds": len(r.detection.detections),
        },
        "segmentation": {
            "total_seconds_scanned": r.segmentation.total_seconds_scanned,
            "seconds_with_detection": r.segmentation.seconds_with_detection,
            "segments": [
                {
                    "index": s.index,
                    "start_seconds": s.start_seconds,
                    "end_seconds": s.end_seconds,
                    "duration_seconds": s.duration_seconds,
                    "start_time": s.start_time,
                    "end_time": s.end_time,
                    "bbox": s.bbox,
                }
                for s in r.segmentation.segments
            ],
        },
        "crop": {
            "output_dir": r.crop.output_dir,
            "segments": [
                {
                    "index": cs.index,
                    "file_path": cs.file_path,
                    "file_name": cs.file_name,
                    "start_seconds": cs.start_seconds,
                    "end_seconds": cs.end_seconds,
                    "duration_seconds": cs.duration_seconds,
                    "bbox": cs.bbox,
                }
                for cs in r.crop.segments
            ],
        },
    }


def deserialize_pipeline_result(job_id: str, data: dict) -> PipelineResult:
    meta = data["detection"]["metadata"]
    return PipelineResult(
        job_id=job_id,
        video_path=data["video_path"],
        output_dir=data["output_dir"],
        detection_times_path=data["detection_times_path"],
        detection=DetectionResult(
            metadata=VideoMetadata(
                fps=meta["fps"],
                frame_count=meta["frame_count"],
                duration_seconds=meta["duration_seconds"],
            ),
        ),
        segmentation=SegmentationResult(
            total_seconds_scanned=data["segmentation"]["total_seconds_scanned"],
            seconds_with_detection=data["segmentation"]["seconds_with_detection"],
            segments=[
                Segment(**s) for s in data["segmentation"]["segments"]
            ],
        ),
        crop=CropResult(
            output_dir=data["crop"]["output_dir"],
            segments=[
                CroppedSegment(**cs) for cs in data["crop"]["segments"]
            ],
        ),
```
